# Run python database/dbManager.py once before the other modules to apply the
# numbered SQL files in database/schemaMigrations (named NNNNdescriptiveName.sql,
# e.g. 0002addRangerAlerts.sql). Migration files must not contain BEGIN/COMMIT;
# each file is wrapped in its own exclusive transaction and failures roll back.
# Any process writing to geometry tables (e.g. acoustic_events) must load SpatiaLite.

# database/schema.sql (species, telemetry, sensor_readings, detections,
# genetics_records, risk_assessments) is applied automatically by dbManager.py
# before numbered migrations; it re-runs only when its checksum changes. Changes to
# columns of existing tables must be added as new schemaMigrations files.
# Requires SQLite 3.38+ for STRICT tables and built-in JSON (check with:
# python3 -c "import sqlite3; print(sqlite3.sqlite_version)"). Foreign keys are
# enforced on dbManager connections; other raw writers must run PRAGMA foreign_keys = ON.
# No additional pip packages are required for the schema.

# api/websocketManager.py - real-time alert fan-out: stateful connection registry, zone targeting, heartbeat, telemetry bridge
pip install "uvicorn[standard]" fastapi h3 websockets
# uvicorn[standard] already pulls in websockets; this just confirms it. Settings live under websocketApi: in config.yaml.
# Alerts are fanned out in-process from the telemetry broadcaster (no Redis needed on the Pi); the same event that
# feeds the SSE stream is translated into a priority-ranked alert and pushed to every matching socket.
# Socket URL: ws://<pi-ip>:8000/api/v1/ws/alerts/<client_id>
# Query parameters: ?token=<ICMIS_API_KEY> &client_type=web|ios|android|field|service &zone=<h3 cell> &min_priority=HIGH
# CLIENT CONTRACT: the server sends {"type":"ping"} every 30s and the client MUST reply {"type":"pong"};
# a socket that misses two pings in a row is treated as dead and flushed.
# Control frames the client may send: {"type":"subscribe","zones":["87..."],"min_priority":"HIGH"},
# {"type":"unsubscribe"}, {"type":"status"}, {"type":"ping"}
# Browser example:
# const ws = new WebSocket("ws://<pi-ip>:8000/api/v1/ws/alerts/dashboard1?client_type=web&min_priority=HIGH");
# ws.onmessage = (e) => { const f = JSON.parse(e.data); if (f.type === "ping") ws.send('{"type":"pong"}'); else console.log(f); };
# Publish an alert from any other service:
# curl -X POST http://<pi-ip>:8000/api/v1/ws/alerts -H "Content-Type: application/json" -d "{\"priority\":\"CRITICAL\",\"event_type\":\"acoustic_anomaly\",\"classification\":\"gunshot_detected\",\"metrics\":{\"cri_score\":88,\"confidence\":0.94},\"location\":{\"lat\":-2.334,\"lng\":34.821}}"
# curl "http://<pi-ip>:8000/api/v1/ws/status?include_connections=true"
# In-process demo: python api/websocketManager.py
